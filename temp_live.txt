import React, { useEffect, useState, useRef } from "react";
import { useParams, useNavigate } from "react-router-dom";
import { Send, ShoppingBag, X, User, Heart, Play, CheckCircle2 } from "lucide-react";
import { v4 as uuidv4 } from "uuid";

interface LiveData {
  id: string;
  title: string;
  productName: string;
  productPrice: number;
  productImage: string;
  videoUrl: string;
}

interface EventData {
  id: string;
  timeSeconds: number;
  message: string;
  buyerName: string;
}

interface MessageData {
  id: string;
  senderName: string;
  message: string;
  senderType: string;
}

export default function LivePage() {
  const { slug } = useParams();
  const navigate = useNavigate();
  const [live, setLive] = useState<LiveData | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [events, setEvents] = useState<EventData[]>([]);
  const [chat, setChat] = useState<MessageData[]>([]);
  const [sessionId] = useState(() => localStorage.getItem("sessionId") || uuidv4());
  
  const [visitorName, setVisitorName] = useState(localStorage.getItem("visitorName") || "");
  const [activeEvent, setActiveEvent] = useState<EventData | null>(null);
  const [chatInput, setChatInput] = useState("");
  const [showCart, setShowCart] = useState(false);
  const [cartItems, setCartItems] = useState<{name: string, price: number, qty: number}[]>([]);
  const [hearts, setHearts] = useState<{id: string, left: number}[]>([]);
  const [isPlaying, setIsPlaying] = useState(false);
  const [viewers, setViewers] = useState(Math.floor(Math.random() * 300) + 1200);
  
  const videoRef = useRef<HTMLVideoElement>(null);
  const chatEndRef = useRef<HTMLDivElement>(null);
  const triggeredEvents = useRef<Set<string>>(new Set());

  const [checkoutStep, setCheckoutStep] = useState<'cart' | 'checkout' | 'success'>('cart');
  const [checkoutData, setCheckoutData] = useState({ name: '', address: '', card: '' });

  useEffect(() => {
    localStorage.setItem("sessionId", sessionId);
    
    fetch(`/api/public/live/${slug}`)
      .then(res => {
        if (!res.ok) throw new Error("Not found");
        return res.json();
      })
      .then(data => {
        if (data.live) {
          setLive(data.live);
          setEvents(data.events || []);
          loadChat(data.live.id);
          
          if (!visitorName) {
            setTimeout(() => askForName(data.live.id), 10000);
          }
        } else {
          setError("Live não encontrada");
        }
      })
      .catch(err => {
        console.error(err);
        setError("Erro ao carregar Live");
      });
  }, [slug, sessionId]);

  useEffect(() => {
    const interval = setInterval(() => {
      setViewers(prev => prev + Math.floor(Math.random() * 7) - 3);
    }, 3000);
    return () => clearInterval(interval);
  }, []);

  const loadChat = async (liveId: string) => {
    const res = await fetch(`/api/public/chat/${liveId}`);
    const data = await res.json();
    setChat(data);
  };

  const askForName = (liveId: string) => {
    const name = window.prompt("Como gostaria de ser chamado no chat?");
    if (name) {
      setVisitorName(name);
      localStorage.setItem("visitorName", name);
      fetch("/api/public/chat", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ liveId, sessionId, message: `👋 ${name} entrou na live!` })
      }).then(() => loadChat(liveId));
    }
  };

  useEffect(() => {
    chatEndRef.current?.scrollIntoView({ behavior: "smooth" });
  }, [chat]);

  const handleTimeUpdate = () => {
    if (!videoRef.current || !live) return;
    const time = Math.floor(videoRef.current.currentTime);
    
    const event = events.find(e => time >= e.timeSeconds && time <= e.timeSeconds + 7);
    if (event && activeEvent?.id !== event.id) {
      setActiveEvent(event);
    } else if (!event && activeEvent) {
      setActiveEvent(null);
    }
  };

  const handleSendMessage = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!chatInput.trim() || !live) return;
    
    let name = visitorName;
    if (!name) {
      name = window.prompt("Como gostaria de ser chamado?") || "Visitante";
      setVisitorName(name);
      localStorage.setItem("visitorName", name);
    }
    
    const msg = chatInput;
    setChatInput("");
    
    setChat(prev => [...prev, { id: Date.now().toString(), senderName: name, message: msg, senderType: "viewer" }]);
    
    await fetch("/api/public/chat", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ liveId: live.id, sessionId, message: msg })
    });
    
    setTimeout(() => loadChat(live.id), 2000);
  };

  const togglePlay = () => {
    if (videoRef.current) {
      if (isPlaying) videoRef.current.pause();
      else videoRef.current.play();
      setIsPlaying(!isPlaying);
    }
  };

  const handleLike = () => {
    const newHeart = { id: uuidv4(), left: Math.random() * 40 + 10 }; 
    setHearts(prev => [...prev, newHeart]);
    setTimeout(() => {
      setHearts(prev => prev.filter(h => h.id !== newHeart.id));
    }, 1500);
  };

  const addToCart = () => {
    if (live) {
      setCartItems(prev => {
        const exists = prev.find(i => i.name === live.productName);
        if (exists) return prev.map(i => i.name === live.productName ? {...i, qty: i.qty + 1} : i);
        return [...prev, { name: live.productName, price: live.productPrice, qty: 1 }];
      });
      setCheckoutStep('cart');
    }
  };

  const processPayment = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!live || cartItems.length === 0) return;
    
    await fetch("/api/public/order", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ liveId: live.id, sessionId, quantity: cartItems[0].qty, unitPrice: cartItems[0].price })
    });
    
    setCheckoutStep('success');
  };

  const closeCart = () => {
    setShowCart(false);
    setTimeout(() => {
      setCheckoutStep('cart');
      if (checkoutStep === 'success') setCartItems([]);
    }, 300);
  };

  if (error) return <div className="h-screen bg-[#111111] text-white flex justify-center items-center text-xl font-bold">{error}</div>;
  if (!live) return <div className="h-screen bg-[#111111] text-white flex justify-center items-center">Carregando...</div>;

  return (
    <div className="min-h-screen bg-[#111111] text-white flex flex-col font-sans overflow-hidden relative">
      
      {/* Top App Bar */}
      <header className="px-6 py-5 flex items-center justify-between border-b border-white/5 relative z-20 bg-[#111111]">
         <div className="flex items-center gap-2 cursor-pointer" onClick={() => navigate('/')}>
          <div className="w-9 h-9 bg-[#FF5A36] rounded-full flex items-center justify-center font-bold text-white text-lg">V</div>
          <span className="text-xl font-bold tracking-tight hidden sm:block">Viva<span className="text-[#FF5A36]">Shop</span></span>
        </div>
        
        <div className="flex items-center gap-4">
           <button className="bg-white text-black px-6 py-2.5 rounded-full font-bold text-sm hidden sm:block">Assistir</button>
           <button onClick={() => navigate('/login')} className="text-gray-400 font-medium text-sm hover:text-white transition">Criar live</button>
           <button onClick={() => setShowCart(true)} className="w-10 h-10 rounded-full bg-white/5 flex items-center justify-center hover:bg-white/10 transition border border-white/10 relative">
             <ShoppingBag className="w-4 h-4 text-gray-300" />
             {cartItems.length > 0 && (
                <span className="absolute -top-1 -right-1 bg-[#FF5A36] text-white text-[10px] font-bold w-4 h-4 flex items-center justify-center rounded-full border-2 border-[#111111]">
                  {cartItems.length}
                </span>
             )}
           </button>
        </div>
      </header>

      {/* Main Content Area - Mobile Centered */}
      <div className="flex-1 flex justify-center items-center p-4 sm:p-8 overflow-hidden">
        <div className="relative w-full max-w-[420px] aspect-[9/16] max-h-[85vh] bg-[#1a1a1a] rounded-[2.5rem] overflow-hidden shadow-2xl flex flex-col mx-auto">
          
          {/* VIDEO PLAYER */}
          <video 
            ref={videoRef}
            src={live.videoUrl} 
            className="absolute inset-0 w-full h-full object-cover"
            onTimeUpdate={handleTimeUpdate}
            onPlay={() => setIsPlaying(true)}
            onPause={() => setIsPlaying(false)}
            onClick={togglePlay}
            playsInline
            loop
            autoPlay
          />

          {/* OVERLAYS CONTAINER */}
          <div className="absolute inset-0 z-10 flex flex-col justify-between pointer-events-none">
            
            {/* Top Overlay - Info */}
            <div className="p-5 pt-8 bg-gradient-to-b from-black/80 to-transparent pointer-events-auto">
               <div className="flex justify-between items-start mb-4">
                 <div className="flex items-center gap-3">
                   <div className="w-12 h-12 rounded-full border-2 border-white/20 overflow-hidden shrink-0 shadow-lg">
                      <img src={`https://api.dicebear.com/7.x/avataaars/svg?seed=${live.id}`} className="w-full h-full object-cover bg-gray-200" alt="Avatar" />
                   </div>
                   <div className="flex flex-col drop-shadow-md">
                     <span className="font-bold text-white leading-tight text-[15px]">Loja Oficial</span>
                     <span className="text-white/80 text-[11px] font-medium">Marina Nativa • {(viewers / 1000).toFixed(1)} mil assistindo</span>
                   </div>
                 </div>
                 <button className="bg-[#FF5A36] text-white font-bold px-4 py-1.5 rounded-full text-sm hover:bg-[#e04825] transition shadow-lg shrink-0">
                   Seguir
                 </button>
               </div>
               
               <h1 className="text-lg font-bold text-white mb-1 shadow-black/50 drop-shadow-md leading-tight max-w-[90%]">{live.title}</h1>
               <p className="text-white/70 text-xs shadow-black/50 drop-shadow-md font-medium">Apresentação ao vivo • replay disponível</p>
            </div>

            {/* Play Button Alert if paused */}
            {!isPlaying && (
              <div className="absolute inset-0 flex items-center justify-center pointer-events-none z-10">
                <div className="bg-black/40 rounded-full p-4 backdrop-blur-md border border-white/10">
                  <Play className="w-12 h-12 text-white/80 ml-1" />
                </div>
              </div>
            )}

            {/* Event Popup Overlay */}
            {activeEvent && (
              <div className="absolute top-1/2 -translate-y-1/2 left-0 right-0 z-20 pointer-events-auto px-4 flex justify-center animate-in slide-in-from-bottom-4 fade-in duration-300">
                <div 
                  className="bg-black/60 backdrop-blur-xl rounded-2xl p-3 shadow-2xl flex items-center gap-3 border border-white/10 cursor-pointer max-w-[300px] w-full"
                  onClick={() => setShowCart(true)}
                >
                  <div className="w-12 h-12 bg-white rounded-xl flex items-center justify-center shrink-0 border border-white/10">
                     <ShoppingBag className="w-6 h-6 text-[#FF5A36]" />
                  </div>
                  <div className="flex-1 min-w-0">
                    <p className="text-[10px] font-bold text-[#FF5A36] uppercase tracking-wider">{activeEvent.buyerName} comprou!</p>
                    <p className="text-xs text-white font-medium truncate">{activeEvent.message}</p>
                  </div>
                </div>
              </div>
            )}

            {/* Bottom Overlay Area */}
            <div className="flex flex-col justify-end p-4 pb-6 bg-gradient-to-t from-black/90 via-black/40 to-transparent">
               
               {/* Chat Bubbles */}
               <div className="flex flex-col gap-2 max-h-[220px] overflow-y-auto no-scrollbar pointer-events-auto chat-mask mb-4">
                 {chat.map(msg => (
                    <div key={msg.id} className="bg-black/40 backdrop-blur-md rounded-[1.25rem] rounded-bl-none px-3.5 py-1.5 self-start max-w-[85%] border border-white/5 flex gap-1.5 items-baseline">
                       <span className="font-bold text-white text-[13px] shrink-0">{msg.senderName}</span>
                       <span className="text-white/90 text-[13px] leading-snug">{msg.message}</span>
                    </div>
                 ))}
                 <div ref={chatEndRef} />
               </div>

               {/* Product Card */}
               <div className="pointer-events-auto bg-black/60 backdrop-blur-xl rounded-[2rem] p-2.5 flex items-center gap-3 border border-white/10 shadow-2xl relative mb-4">
                  <div className="w-14 h-14 bg-white rounded-2xl overflow-hidden shrink-0 border border-white/10 flex items-center justify-center p-1">
                     {getDisplayProduct().image ? (
                        <img src={getDisplayProduct().image} className="w-full h-full object-contain p-1" />
                      ) : (
                        <ShoppingBag className="text-gray-300 w-8 h-8"/>
                      )}
                    </div>
                    <div className="flex-1 flex flex-col justify-between">
                      <div>
                        <div className="flex justify-between items-start">
                          <h4 className="font-bold text-white text-sm leading-tight">{getDisplayProduct().name}</h4>
                        </div>
                        <div className="flex items-baseline gap-2 mt-2">
                          <span className="font-black text-[#FF5A36] text-lg">R$ {getDisplayProduct().price?.toFixed(2)}</span>
                          <span className="text-[10px] text-gray-500 line-through">R$ {(getDisplayProduct().price * 1.5).toFixed(2)}</span>
                        </div>
                      </div>
                      <div className="flex justify-between items-center mt-2">
                        <span className="text-gray-400 text-xs">Quantidade: 1</span>
                        <button 
                          onClick={addToCart}
                          className="text-[#FF5A36] text-xs font-bold px-3 py-1 bg-[#FF5A36]/10 rounded-full hover:bg-[#FF5A36]/20 transition"
                        >
                          + Adicionar
                        </button>
                      </div>
                    </div>
                  </div>

                  {/* Cart Content */}
                  {cartItems.length > 0 && (
                    <div className="bg-white/5 border border-white/10 rounded-2xl p-4">
                      <h4 className="font-bold text-white mb-4 text-sm uppercase tracking-wider">Resumo da Sacola</h4>
                      <div className="space-y-3">
                        {cartItems.map((item, idx) => (
                          <div key={idx} className="flex justify-between items-center text-sm">
                            <span className="font-medium text-gray-300 flex items-center gap-2">
                              <span className="bg-white/10 px-2 py-0.5 rounded text-xs">{item.qty}x</span> 
                              {item.name}
                            </span>
                            <span className="font-bold text-white">R$ {(item.price * item.qty).toFixed(2)}</span>
                          </div>
                        ))}
                        <div className="pt-3 mt-3 border-t border-white/10 flex justify-between items-center">
                          <span className="font-bold text-gray-400 text-sm">Total</span>
                          <span className="font-black text-[#FF5A36] text-lg">
                            R$ {cartItems.reduce((a,c) => a + c.price * c.qty, 0).toFixed(2)}
                          </span>
                        </div>
                      </div>
                    </div>
                  )}
                </div>

                <div className="p-6 border-t border-white/10 bg-[#1A1A1A] pb-safe">
                  <button 
                    onClick={() => setCheckoutStep('checkout')}
                    disabled={cartItems.length === 0}
                    className="w-full bg-[#FF5A36] disabled:bg-gray-700 disabled:text-gray-500 text-white font-bold py-4 rounded-full uppercase tracking-widest text-sm active:scale-95 transition-all shadow-[0_0_20px_rgba(255,90,54,0.3)] disabled:shadow-none"
                  >
                    Ir para Pagamento
                  </button>
                </div>
              </>
            )}

            {/* Step 2: Checkout Form */}
            {checkoutStep === 'checkout' && (
              <form onSubmit={processPayment} className="flex-1 overflow-y-auto flex flex-col">
                <div className="p-6 space-y-5 flex-1">
                  <div>
                    <label className="block text-xs font-bold text-gray-400 mb-1.5 uppercase tracking-wider">Nome Completo</label>
                    <input required type="text" className="w-full bg-black/50 border border-white/10 text-white rounded-xl px-4 py-3 outline-none focus:border-[#FF5A36] transition-colors" value={checkoutData.name} onChange={e => setCheckoutData({...checkoutData, name: e.target.value})} placeholder="Seu nome" />
                  </div>
                  <div>
                    <label className="block text-xs font-bold text-gray-400 mb-1.5 uppercase tracking-wider">Endereço de Entrega</label>
                    <input required type="text" className="w-full bg-black/50 border border-white/10 text-white rounded-xl px-4 py-3 outline-none focus:border-[#FF5A36] transition-colors" value={checkoutData.address} onChange={e => setCheckoutData({...checkoutData, address: e.target.value})} placeholder="Sua rua, número" />
                  </div>
                  <div>
                    <label className="block text-xs font-bold text-gray-400 mb-1.5 uppercase tracking-wider">Cartão de Crédito</label>
                    <input required type="text" placeholder="0000 0000 0000 0000" className="w-full bg-black/50 border border-white/10 text-white rounded-xl px-4 py-3 outline-none focus:border-[#FF5A36] transition-colors font-mono" value={checkoutData.card} onChange={e => setCheckoutData({...checkoutData, card: e.target.value})} />
                  </div>
                </div>
                <div className="p-6 border-t border-white/10 bg-[#1A1A1A] pb-safe">
                  <button type="submit" className="w-full bg-[#FF5A36] text-white font-bold py-4 rounded-full uppercase tracking-widest text-sm active:scale-95 transition-all shadow-[0_0_20px_rgba(255,90,54,0.3)]">
                    Confirmar Pedido (R$ {cartItems.reduce((a,c) => a + c.price * c.qty, 0).toFixed(2)})
                  </button>
                </div>
              </form>
            )}

            {/* Step 3: Success */}
            {checkoutStep === 'success' && (
              <div className="flex-1 p-8 flex flex-col items-center justify-center text-center pb-safe">
                <div className="w-20 h-20 bg-green-500/20 rounded-full flex items-center justify-center mb-6">
                  <CheckCircle2 className="w-10 h-10 text-green-500" />
                </div>
                <h3 className="text-2xl font-bold text-white mb-2">Pedido Confirmado!</h3>
                <p className="text-gray-400 text-sm mb-8 leading-relaxed max-w-[250px]">Seu pagamento foi aprovado e o vendedor já foi notificado.</p>
                <button onClick={closeCart} className="w-full bg-white text-black font-bold py-4 rounded-full uppercase tracking-widest text-sm active:scale-95 transition-transform">
                  Voltar para a Live
                </button>
              </div>
            )}

          </div>
        </div>
      )}
    </div>
  );
}
